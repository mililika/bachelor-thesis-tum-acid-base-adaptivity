\newacronym{cpr}{CPR}{Consistency Preservation Rule}
\newacronym{ewma}{EWMA}{exponentially weighted moving average}
\newacronym{c2pl}{C2PL}{conservative two-phase locking}
\newacronym{tps}{tps}{transactions per second}
\newacronym{tuma}{TUM}{Technical University of Munich}

\newglossaryentry{dirtyresource}
{
  name=dirty resource,
  description={a resource carrying a consistency-preserving update that
               has not yet been applied}
}

\newglossaryentry{reconciliation}
{
  name=reconciliation,
  description={the background work that applies pending
               consistency-preserving updates and clears dirty resources}
}

\newglossaryentry{semanticoverlap}
{
  name=semantic overlap,
  description={the degree to which two models represent the same
               underlying information}
}

\newglossaryentry{modeoracle}
{
  name=mode oracle,
  description={the component that decides, at admission of a root
               transaction, whether it executes under ACID or BASE
               semantics}
}
